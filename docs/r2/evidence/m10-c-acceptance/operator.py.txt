"""M10-C live operator acceptance on the Quality page: every step is a click or a typed value in a visible Chrome.

Usage: uv run python operator.py <run_dir> <step>...   (steps: profile probes review seal decide package axe)
Saves screenshots and the visible step text under <run_dir>/shots, and the axe result in <run_dir>/axe.json.
"""
import asyncio
import json
import re
import sys
from pathlib import Path

from playwright.async_api import Page, async_playwright

RUN = Path(sys.argv[1]).resolve()
STEPS = sys.argv[2:]
BASE = "http://127.0.0.1:5174"
SHOTS = RUN / "shots"
SHOTS.mkdir(exist_ok=True)
LOG: list[dict] = []


async def shot(page: Page, name: str) -> None:
    await page.screenshot(path=str(SHOTS / f"{name}.png"), full_page=True)


async def section(page: Page, kicker: str) -> str:
    return await page.locator("section", has=page.locator(f"h2:text-is('{kicker}')")).inner_text()


async def open_quality(page: Page) -> None:
    password = (RUN / "auth-password").read_text("utf-8").strip()
    await page.goto(f"{BASE}/login")
    await page.locator("input").nth(0).fill("actor")
    await page.locator("input[type=password]").fill(password)
    await page.locator("button[type=submit]").click()
    await page.wait_for_url(lambda u: "/login" not in u, timeout=30000)
    await page.wait_for_timeout(3000)
    skip = page.get_by_role("button", name="Skip")
    if await skip.count():
        await skip.first.click()
    await page.goto(f"{BASE}/app/projects/project/quality")
    await page.get_by_label("Approved video").wait_for(timeout=30000)
    await page.wait_for_timeout(1500)


async def choose_scope(page: Page) -> None:
    master = page.get_by_label("Approved video")
    await master.select_option(index=1)
    profile = page.get_by_role("combobox", name="Quality profile")
    await profile.locator("option").nth(1).wait_for(state="attached", timeout=15000)
    await profile.select_option(index=1)
    await page.get_by_role("combobox", name="Revision").wait_for(timeout=15000)
    await page.wait_for_timeout(2000)


async def confirm(page: Page, button: str, name: str) -> str:
    group = page.get_by_role("group", name=re.compile("Confirm the prepared request"))
    await group.wait_for(timeout=30000)
    await shot(page, name + "-confirm")
    text = await group.inner_text()
    await group.get_by_role("button", name=button).click()
    await page.wait_for_timeout(1500)
    await group.wait_for(state="detached", timeout=300000)
    await page.wait_for_timeout(2000)
    await shot(page, name + "-done")
    return text


async def profile(page: Page) -> None:
    await page.get_by_label("Approved video").select_option(index=1)
    for label, value in (
        ("Platform profile", "youtube-long"),
        ("Width (px)", "1920"),
        ("Height (px)", "1080"),
        ("Minimum duration (ms)", "60000"),
        ("Maximum duration (ms)", "900000"),
        ("Required locales (comma separated)", "en"),
    ):
        await page.get_by_label(label).fill(value)
    await page.get_by_role("button", name="Load template").click()
    await page.get_by_label(re.compile("Quality policy")).wait_for(timeout=15000)
    await page.get_by_role("button", name=re.compile("^Prepare publication$")).click()
    LOG.append({"step": "profile", "confirm": await confirm(page, "Publish profile", "10-profile")})


async def probes(page: Page) -> None:
    await choose_scope(page)
    for button, name in (("Check container and streams", "20-container"), ("Check full decode", "21-decode")):
        await page.get_by_role("button", name=button).click()
        LOG.append({"step": name, "confirm": await confirm(page, "Run check", name)})
    LOG.append({"step": "probes", "section": await section(page, "PROBES")})


async def review(page: Page) -> None:
    await choose_scope(page)
    await page.get_by_label(re.compile("^Your notes")).fill(
        "Watched the full 75 s explainer: the narration matches the three shots, captions are readable, audio is clean."
    )
    await page.get_by_role("button", name="Prepare review").click()
    LOG.append({"step": "review", "confirm": await confirm(page, "Record review", "30-review")})


async def seal(page: Page) -> None:
    await choose_scope(page)
    box = page.locator("section", has=page.locator("h2:text-is('SEAL')"))
    for requirement in ("technical-container", "technical-decode", "review"):
        await box.locator(f'input[name="qa-seal-{requirement}"]').first.check()
    await shot(page, "40-seal-chosen")
    await page.get_by_role("button", name="Prepare run").click()
    LOG.append({"step": "seal", "confirm": await confirm(page, "Seal run", "40-seal")})
    LOG.append({"step": "runs", "section": await section(page, "RUNS")})


async def decide(page: Page) -> None:
    await choose_scope(page)
    runs = page.locator("section", has=page.locator("h2:text-is('RUNS')"))
    await runs.locator('input[name="qa-run"]').first.check()
    await page.wait_for_timeout(1500)
    await page.get_by_label("Decision").select_option("PASS")
    await page.get_by_label("Reason").fill("Both probes pass and the Human review found no defect.")
    await page.get_by_role("button", name="Prepare decision").click()
    LOG.append({"step": "decide", "confirm": await confirm(page, "Issue decision", "50-decide")})
    LOG.append({"step": "runs-after", "section": await section(page, "RUNS")})


async def package(page: Page) -> None:
    await choose_scope(page)
    link = page.get_by_role("link", name="Package this run")
    await link.first.wait_for(timeout=20000)
    LOG.append({"step": "package", "href": await link.first.get_attribute("href")})
    await link.first.click()
    await page.wait_for_timeout(4000)
    await shot(page, "60-packaging-prefilled")
    LOG.append({"step": "packaging", "url": page.url, "run_field": await page.get_by_label(re.compile("^Quality run ID")).input_value()})


async def axe(page: Page) -> None:
    await choose_scope(page)
    await page.locator('section', has=page.locator("h2:text-is('RUNS')")).locator('input[name="qa-run"]').first.check()
    await page.wait_for_timeout(1500)
    await page.get_by_label("Decision").select_option("WAIVE")
    await page.add_script_tag(path="/home/anhtuan/content-production-os/.worktrees/r2-m10-c/frontend/node_modules/.pnpm/axe-core@4.13.0/node_modules/axe-core/axe.min.js")
    result = await page.evaluate("async () => { const r = await axe.run(document.querySelector('main') || document.body); return r.violations.map(v => ({id: v.id, impact: v.impact, nodes: v.nodes.length, help: v.help})); }")
    (RUN / "axe.json").write_text(json.dumps(result, indent=2), encoding="utf-8")
    LOG.append({"step": "axe", "violations": result})


async def reload(page: Page) -> None:
    await page.reload()
    await page.get_by_label("Approved video").wait_for(timeout=30000)
    await choose_scope(page)
    await shot(page, "70-after-reload")
    LOG.append({"step": "reload", "runs": await section(page, "RUNS")})


async def main() -> None:
    async with async_playwright() as p:
        browser = await p.chromium.launch(channel="chrome", headless=False, slow_mo=40)
        page = await (await browser.new_context(locale="en-US", viewport={"width": 1440, "height": 1000})).new_page()
        page.on("dialog", lambda dialog: asyncio.ensure_future(dialog.dismiss()))
        await open_quality(page)
        try:
            for step in STEPS:
                await globals()[step](page)
        finally:
            await shot(page, "zz-last")
            out = RUN / "operator.json"
            prior = json.loads(out.read_text("utf-8")) if out.exists() else []
            out.write_text(json.dumps(prior + LOG, ensure_ascii=False, indent=2), encoding="utf-8")
            alerts = await page.get_by_role("alert").all_inner_texts()
            print(json.dumps({"log": LOG, "alerts": alerts}, ensure_ascii=False, indent=1)[:6000])
            await browser.close()


asyncio.run(main())
