"""Live accessibility check of the Packaging page in Chrome: axe-core with every rule (contrast included) on the
empty and the populated refusal/history state, plus a keyboard Tab traversal with visible focus."""
import asyncio, json, re, sys
from pathlib import Path
from playwright.async_api import async_playwright
AXE = Path("/home/anhtuan/content-production-os/.worktrees/r2-m10-b/frontend/node_modules/axe-core/axe.min.js").read_text()
async def axe(page, selector):
    await page.add_script_tag(content=AXE)
    result = await page.evaluate("async (sel) => { const r = await axe.run(document.querySelector(sel)); return r.violations.map(v => ({id: v.id, impact: v.impact, nodes: v.nodes.length, sample: v.nodes.slice(0,3).map(n => n.target.join(' '))})) }", selector)
    return result
async def main(pilot, cohort, out):
    state = json.loads((pilot / "state.json").read_text()); pw = (pilot / "auth-password").read_text().strip()
    res = {}
    async with async_playwright() as p:
        b = await p.chromium.launch(channel="chrome", headless=False)
        c = await b.new_context(locale="en-US", viewport={"width": 1440, "height": 1000}, color_scheme="dark")
        page = await c.new_page()
        await page.goto("http://127.0.0.1:5173/login")
        await page.locator("input").nth(0).fill("actor"); await page.locator("input[type=password]").fill(pw)
        await page.locator("button[type=submit]").click()
        await page.wait_for_url(re.compile(r"/app/projects"), timeout=30000); await page.wait_for_timeout(3000)
        skip = page.get_by_role("button", name="Skip")
        if await skip.count(): await skip.first.click()
        await page.goto("http://127.0.0.1:5173/app/projects/project/packaging")
        await page.get_by_label(re.compile("^Quality run ID")).wait_for(timeout=30000)
        main_sel = "main" if await page.locator("main").count() else "body"
        res["empty"] = await axe(page, main_sel)
        await page.get_by_label(re.compile("^Video artifact key")).fill(state["root_key"])
        await page.get_by_label(re.compile("^Cohort")).fill("cohort")
        await page.get_by_label(re.compile("^Product surface")).fill("video")
        await page.get_by_text("Exported archives").wait_for(timeout=30000)
        await page.screenshot(path=str(pilot / "shots" / "a11y-refusal-history.png"), full_page=True)
        res["refusal_history_text"] = (await page.locator("section[aria-labelledby=pk-export]").inner_text())[:600]
        res["refusal_history"] = await axe(page, main_sel)
        await page.get_by_label(re.compile("^Quality run ID")).focus()
        seen = []
        for _ in range(40):
            await page.keyboard.press("Tab")
            info = await page.evaluate("""() => { const e = document.activeElement; const s = getComputedStyle(e);
                return {tag: e.tagName, label: (e.getAttribute('aria-label') || e.labels?.[0]?.innerText || e.innerText || '').slice(0,40),
                        outline: s.outlineStyle + ' ' + s.outlineWidth, shadow: s.boxShadow !== 'none', disabled: !!e.disabled} }""")
            seen.append(info)
        res["tab_order"] = seen
        res["focus_visible_all"] = all(i["outline"].split()[0] != "none" or i["shadow"] for i in seen if i["tag"] in ("INPUT","TEXTAREA","BUTTON","A"))
        await b.close()
    Path(out).write_text(json.dumps(res, indent=2, ensure_ascii=False)); print(json.dumps({k: v for k, v in res.items() if k != "tab_order"}, indent=1, ensure_ascii=False)[:3000])
asyncio.run(main(Path(sys.argv[1]), sys.argv[2], sys.argv[3]))
