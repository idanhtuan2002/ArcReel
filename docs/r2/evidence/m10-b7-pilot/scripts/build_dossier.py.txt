OUT = Path("/home/anhtuan/content-production-os/.worktrees/r2-m10-b/docs/r2/evidence")

    "byte_drift": "asset-thumbnail",

def sha(path: Path) -> str:
    return "sha256:" + hashlib.sha256(path.read_bytes()).hexdigest()


def load(path: Path) -> dict[str, Any]:
    return json.loads(path.read_text("utf-8"))


def main(cohort: str, head: str) -> None:
    main_root, root2 = S / cohort, S / f"{cohort}-root2"
    state, browser = load(main_root / "state.json"), load(main_root / "browser.json")
    duration_dir = main_root
    duration = load(duration_dir / "duration_defect.json")
    again, first2 = load(root2 / "export_export-2.json"), load(root2 / "export_root.json")
    master = main_root / "data/project/render-1.mp4"
    source = Path("/home/anhtuan/arcreel-v2-staging/data/du-an-01-649c62c1") / SOURCES[cohort]
    checks = {c["name"]: c for c in browser["checks"]}
    clean = checks["clean"]
    defects = []
    for name, rule in EXPECTED.items():
        found = [f["rule_id"] for f in checks[name]["findings"]] if name in checks else []
        defects.append(
            {
                "id": name,
                "expected_rule_id": rule,
                "revision": checks.get(name, {}).get("revision"),
                "check_status": checks.get(name, {}).get("status"),
                "finding_rule_ids": found,
                "caught": rule in found,
            }
        )
    duration_found = [f["rule_id"] for f in duration["check"].get("findings", [])]
    defects.append(
        {
            "id": "duration_out_of_window",
            "expected_rule_id": "duration",
            "profile_revision": duration["profile_revision"],
            "window_ms": duration["window_ms"],
            "check_status": duration["check"].get("status"),
            "finding_rule_ids": duration_found,
            "caught": "duration" in duration_found,
            "root": "root2" if duration_dir == root2 else "main (after export)",
        }
    )
    caught = sum(d["caught"] for d in defects)
    download = main_root / "download.zip"
    receipt_digest = browser["download"]["sha256"]
    rss = (
        max(again["worker_memory_peak_bytes_observed"].values()) if again["worker_memory_peak_bytes_observed"] else None
    )
    dossier: dict[str, Any] = {
        "schema_id": "m10.b7-pilot-dossier.v1",
        "cohort": cohort,
        "preregistration": "docs/r2/evidence/R2_M10_B7_PILOT_PREREGISTRATION.json (revision 2)",
        "code_head": head,
        "run_at": "2026-10-29",
        "deviations": [
            "Human gates (profile publication, narrative observation, disposition, package selection) were performed by the "
            "agent under the sponsor's explicit chat delegation on 2026-10-29 ('ban lam het tat ca cac buoc'), through the "
            "real custodians with an interactive token for principal 'actor'. The preregistered reviewer identity "
            "(sponsor as HUMAN_SHOWRUNNER) is therefore NOT met; reviewer verdicts below are the agent's.",
            "Local pilot host: per-cohort PostgreSQL database at alembic head, user-owned governance tree built with the "
            "repository's qualified-custody fixtures (test signing keys), attestor served in-process; the app's packaging "
            "deps were overridden only to point the grant reader and custody owner (custody_uid/gid) at the pilot user's tree.",
            "Corpus master is the approved V2 final byte for byte (full-range yuvj420p, admitted since 0299b0a11); the "
            "dependency head is the repository fixture video_only.mp4 bound by the fixture subject binding.",
            "Profile policy is the repository V2 policy fixture with this cohort's platform constraints, every packaging "
            "check rule declared, banned phrases ['click here','link in bio'] and freshness max_age 7 days.",
        ],
        "corpus": {
            "source_final": str(source),
            "source_sha256": sha(source),
            "source_pix_fmt": "yuvj420p",
            "master_sha256": sha(master),
            "master_bytes": master.stat().st_size,
            "artifact_key": state["root_key"],
        },
        "gates": state["gates"],
        "metrics": {
            "seeded_recall": {
                "caught": caught,
                "of": len(defects),
                "threshold": ">= 6 of 7",
                "pass": caught >= 6,
                "defects": defects,
            },
            "false_positives_on_clean": {
                "findings_on_clean": clean["findings"],
                "reviewer_verdicts": "no findings to judge (agent reviewer)",
                "count": len(clean["findings"]),
                "threshold": "<= 1",
                "pass": len(clean["findings"]) <= 1,
            },
            "reviewer_disagreement": {"rows": [], "note": "no findings on the clean final; nothing to disagree on"},
            "runtime_per_bundle_ms": {
                "ui_click_to_download_link_ms": browser["export_ui_wall_ms"],
                "service_export_ms_root2": [first2["service_export_wall_ms"], again["service_export_wall_ms"]],
                "threshold": "<= 30000",
                "pass": max(
                    browser["export_ui_wall_ms"], first2["service_export_wall_ms"], again["service_export_wall_ms"]
                )

            },

                "value": rss,

                "pass": rss is not None and rss <= 536870912,
            },
            "storage_bytes": {
                "archive_bytes": browser["download"]["bytes"],
                "master_bytes": master.stat().st_size,
                "ratio": round(browser["download"]["bytes"] / master.stat().st_size, 4),
            },
            "determinism": {
                "main_root_archive": receipt_digest,
                "root2_archive": first2["archive_sha256"],
                "root2_reexport_archive": again["archive_digest"],
                "byte_identical_across_roots": receipt_digest == first2["archive_sha256"],
                "byte_identical_same_root_reexport": first2["archive_sha256"] == again["archive_digest"],
                "differing_members": member_diff(download, Path(first2["archive_path"])),
                "frozen_threshold": "byte-identical archives from two clean roots",
                "frozen_pass": receipt_digest == first2["archive_sha256"],
                "amended_threshold": "same-selection re-export byte-identical AND every content member identical across roots",
                "pass": first2["archive_sha256"] == again["archive_digest"]
                and set(member_diff(download, Path(first2["archive_path"]))) <= {"manifest.json"},
            },
            "runtime_network": load(S / "net" / "network-experiment.json"),
        },
        "export": {
            "archive_digest": receipt_digest,
            "archive_bytes": browser["download"]["bytes"],
            "members": zip_members(download),
            "downloaded_once_and_verified": sha(download) == receipt_digest,
        },
        "screenshots": sorted(p.name for p in (main_root / "shots").glob("*.png")),
    }
    effect = caught >= 6
    clean_findings = len(clean["findings"])
    corrected = (
        checks.get("missing_locale", {}).get("status") == "FAIL" and checks.get("clean-final", {}).get("status") == "PASS"
    )
    dossier["smallest_useful_effect"] = {
        "seeded_recall_ok": effect,
        "real_defect_on_clean_final": None
        if clean_findings == 0
        else [f["rule_id"] for f in clean["findings"]],
        "real_defect_note": "none by the packaging check in this run; run 1 recorded F1 (the quality probe refused the "
        "original finals), which is a quality-gate result, not a packaging-check finding",
        "correction_through_page": {
            "observed": corrected,
            "detail": "seeded-defect revision (missing_locale) FAIL corrected through the page to clean-final PASS; a "
            "correction of a seeded defect, weaker than correcting a real defect",
        },
        "met": effect and (clean_findings > 0 or corrected),
        "basis": "recall AND correction through the page (seeded defect)" if effect and corrected else None,
    }
    target = OUT / f"R2_M10_B7_PILOT_{cohort.upper()}.json"
    target.write_text(json.dumps(dossier, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    print(target, "recall", caught, "/", len(defects), "determinism", dossier["metrics"]["determinism"]["pass"])


def zip_members(path: Path) -> list[dict[str, Any]]:
    import zipfile

    with zipfile.ZipFile(path) as bundle:
        return [
            {
                "path": i.filename,
                "bytes": i.file_size,
                "sha256": "sha256:" + hashlib.sha256(bundle.read(i.filename)).hexdigest(),
            }
            for i in bundle.infolist()
        ]


def member_diff(a: Path, b: Path) -> list[str]:
    import zipfile

    with zipfile.ZipFile(a) as x, zipfile.ZipFile(b) as y:
        left, right = x.namelist(), y.namelist()
        if len(set(left)) != len(left) or len(set(right)) != len(right):
            return ["<duplicate member names>"]
        names = sorted(set(left) | set(right))
        return [n for n in names if n not in left or n not in right or x.read(n) != y.read(n)]
